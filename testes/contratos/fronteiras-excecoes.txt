# Contrato K9: exceções do teste de fronteira (testes/fronteiras.sh).
# Formato: CONTRATO ARQUIVO TRECHO. TRECHO é a linha do código sem os espaços
# das pontas. A lista só diminui: quem retira um acoplamento retira a exceção
# no mesmo commit, e o teto de cada contrato fica em testes/fronteiras.sh.
#
# K3: programas do sistema (hyprctl, waybar, notify-send) usados só quando
# existem (command -v), sem erro na ausência.
# K6: comandos do jangada-shell e do jangada-monitor procurados no PATH, com
# command -v, e tratados sem erro na ausência. Nenhum é chamado por
# $JANGADA_CORE_PATH.
K3 bin/jangada-config if command -v waybar >/dev/null 2>&1; then pkill -RTMIN+10 -x waybar 2>/dev/null || true; fi ;;
K3 bin/jangada-config if command -v notify-send >/dev/null 2>&1; then
K3 bin/jangada-config notify-send -a "$app" "${opcoes[@]}" "$1" "$2" 2>/dev/null || true
K3 bin/jangada-config command -v hyprctl >/dev/null 2>&1 || return 0
K3 bin/jangada-config hyprctl activewindow -j 2>/dev/null | jq -r '.title // empty' 2>/dev/null || true
K3 bin/jangada-config if [[ -n "${HYPRLAND_INSTANCE_SIGNATURE:-}" ]] && command -v hyprctl >/dev/null 2>&1; then
K3 bin/jangada-config endereco="$(hyprctl clients -j 2>/dev/null | jq -r --arg s "$1" '(map(select(.title == $s)) + map(select(.title | startswith($s)))) | .[0].address // empty' 2>/dev/null || true)"
K3 bin/jangada-config hyprctl dispatch "hl.dsp.focus({ window = \"address:$endereco\" })" >/dev/null || return 2
K6 bin/jangada-agente if ((snapshot)) && command -v jangada-snapshot >/dev/null 2>&1; then
K6 bin/jangada-agente jangada-snapshot --agente "antes do agente $sessao" || true
K6 bin/jangada-config if command -v jangada-terminal >/dev/null 2>&1; then
K6 bin/jangada-config if ! command -v jangada-tarefas >/dev/null 2>&1; then
K6 bin/jangada-config exec jangada-tarefas "$@"
K6 bin/jangada-validar if ! command -v jangada-subagentes >/dev/null 2>&1; then
K6 bin/jangada-validar resumo="$(timeout 30 jangada-subagentes --entrega "$origem_dir" ${desde:+--desde "$desde"} 2>/dev/null)" || rc=$?
